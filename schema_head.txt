enum RecordStatus {
  DRAFT
  ACTIVE
  INACTIVE
  ARCHIVED
}

enum ApprovalStatus {
  PENDING
  APPROVED
  REJECTED
}

enum ExecutionStatus {
  PENDING
  RUNNING
  COMPLETED
  FAILED
  CANCELLED
}

model RuntimePlugin {
  id           String       @id @default(cuid())
  pluginKey    String       @unique
  name         String
  version      String
  status       RecordStatus @default(ACTIVE)
  capabilities Json
  metadata     Json?
  createdAt    DateTime     @default(now())
  updatedAt    DateTime     @updatedAt

  @@index([status])
}

model Blueprint {
  id           String             @id @default(cuid())
  blueprintKey String             @unique
  name         String
  description  String?
  status       RecordStatus       @default(DRAFT)
  definition   Json
  metadata     Json?
  versions     BlueprintVersion[]
  approvals    ApprovalGate[]
  plans        ExecutionPlan[]
  createdAt    DateTime           @default(now())
  updatedAt    DateTime           @updatedAt

  @@index([status])
}

model BlueprintVersion {
  id          String   @id @default(cuid())
  blueprintId String
  version     Int
  definition  Json
  changeNote  String?
  createdAt   DateTime @default(now())

  blueprint Blueprint @relation(fields: [blueprintId], references: [id], onDelete: Cascade)

  @@unique([blueprintId, version])
  @@index([blueprintId])
}

model ApprovalGate {
  id           String         @id @default(cuid())
  blueprintId  String
  gateType     String
  status       ApprovalStatus @default(PENDING)
  approvedBy   String?
  decisionNote String?
  decidedAt    DateTime?
  createdAt    DateTime       @default(now())
  updatedAt    DateTime       @updatedAt

  blueprint Blueprint @relation(fields: [blueprintId], references: [id], onDelete: Cascade)

  @@index([blueprintId])
  @@index([status])
}

model ExecutionPlan {

  startedAt   DateTime?

  updatedAt   DateTime        @updatedAt

  blueprint Blueprint @relation(fields: [blueprintId], references: [id], onDelete: Cascade)

  @@index([blueprintId])
  @@index([status])
}

model KnowledgeNode {
  id          String              @id @default(cuid())
  nodeKey     String              @unique
  type        String
  name        String
  description String?
  properties  Json
  status      RecordStatus        @default(ACTIVE)
  outgoing    KnowledgeEdge[]     @relation("OutgoingEdges")
  incoming    KnowledgeEdge[]     @relation("IncomingEdges")
  evidence    KnowledgeEvidence[]
  insights    KnowledgeInsight[]
  createdAt   DateTime            @default(now())
  updatedAt   DateTime            @updatedAt

  @@index([type])
  @@index([status])
}

model KnowledgeEdge {
  id         String   @id @default(cuid())
  sourceId   String
  targetId   String
  relation   String
  properties Json?
  createdAt  DateTime @default(now())
  updatedAt  DateTime @updatedAt

  source KnowledgeNode @relation("OutgoingEdges", fields: [sourceId], references: [id], onDelete: Cascade)
  target KnowledgeNode @relation("IncomingEdges", fields: [targetId], references: [id], onDelete: Cascade)

  @@unique([sourceId, targetId, relation])
  @@index([sourceId])
  @@index([targetId])
  @@index([relation])
}

  id                 String       @id @default(cuid())
